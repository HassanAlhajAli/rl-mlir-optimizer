cmake_minimum_required(VERSION 3.22)
project(rl_mlir_optimizer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(MLIR REQUIRED CONFIG PATHS /usr/lib/llvm-18/lib/cmake/mlir)
if(NOT LLVM_PACKAGE_VERSION VERSION_EQUAL "18.1.3")
  message(FATAL_ERROR "The protocol requires LLVM 18.1.3, found ${LLVM_PACKAGE_VERSION}")
endif()
add_executable(rlmlir-worker cpp/compiler.cpp cpp/schedule.cpp cpp/runtime.cpp cpp/cache.cpp cpp/worker.cpp)
target_include_directories(rlmlir-worker SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS})
target_include_directories(rlmlir-worker PRIVATE cpp)
target_compile_definitions(rlmlir-worker PRIVATE ${LLVM_DEFINITIONS})
target_compile_options(rlmlir-worker PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
target_link_libraries(rlmlir-worker PRIVATE MLIRExecutionEngine MLIR LLVM)
set_target_properties(rlmlir-worker PROPERTIES BUILD_RPATH /usr/lib/llvm-18/lib)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME jit_reference COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_jit.py $<TARGET_FILE:rlmlir-worker>)
set_tests_properties(jit_reference PROPERTIES TIMEOUT 120)
add_test(NAME schedules COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_schedules.py $<TARGET_FILE:rlmlir-worker>)
set_tests_properties(schedules PROPERTIES TIMEOUT 180)
add_test(NAME runtime COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_runtime.py $<TARGET_FILE:rlmlir-worker>)
set_tests_properties(runtime PROPERTIES TIMEOUT 180)
